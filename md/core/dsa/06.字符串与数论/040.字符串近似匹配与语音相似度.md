# 概述

实体对齐（record linkage）与模糊查重的主力工具箱：不回答「怎么编辑过去」，而是直接给一个 **[0,1] 相似度分数**。三类六种：**形状类**——Jaro / Jaro-Winkler（匹配字符+换位计数，前缀加权）、Hamming（等长逐位比）、Q-Gram（k 子串重合度，可跨长度）；**复合类**——Monge-Elkan（把编辑距离嵌进「词组对齐」，地址匹配王牌）；**语音类**——Soundex（把姓名**按发音**编码成 4 位码，同音异拼归一）。集合系数的近亲在 [08.机器学习算法/020.集合相似度系数](../08.机器学习算法/020.集合相似度系数.md)。

**本篇收录**：Jaro、Jaro-Winkler、Monge-Elkan、Q-Gram、Soundex、Hamming

- 所属章节：[000.字符串与数论概述](./000.字符串与数论概述.md)

- 复杂度：Hamming O(n)；Jaro/JW O(n·m) 最坏（窗口剪枝后实践近线性）；Q-Gram 建 multiset O(n)、比较 O(q)；Monge-Elkan O(词数² × 编辑距离)；Soundex O(n)。
- 应用场景：客户主数据去重（CRM 里 `Jon Smith / John Smyth`）、电商商品名对齐、地址解析（Monge-Elkan 是 US Census/Google 地址匹配组件）、美国人口普查姓名匹配（Jaro-Winkler 官方阈值 0.85）、DNA/蛋白质序列 motif 比对（Hamming 定长等长场景）、拼写建议候选粗筛、反欺诈（开户姓名音近黑名单库）。

4. Winkler 修正：**公共前缀**（最多 4）每加一乘 0.1 上浮，`JW = J + p·l·(1−J)`，`MAR` 前缀长 3 → `0.944 + 0.1·3·0.056 ≈ 0.961`；
5. 直觉：姓名/短串「开头像更可能真像」——前缀奖励是给人类书写习惯打的补丁。

**动画：Monge-Elkan——「词组各自找对象」的两层匹配**
<!-- dsa.assets/similarity/monge-elkan.gif -->

x=`345 Greystone Court`，y=`345 GreenStone Ct`（字符相似度用归一化编辑距离）：

1. 切词：x→`[345, Greystone, Court]`，y→`[345, GreenStone, Ct]`；
2. 对 x 的**每个词**，在 y 的全部词里挑最像的一个：`345↔345`（1.0）、`Greystone↔GreenStone`（编辑距离 1 → 0.909）、`Court↔Ct`（0.5）；
3. 取这些「最佳伴侣分」的**平均**：`(1.0+0.909+0.5)/3 ≈ 0.803`——空伴侣的词按 0 计，天然惩罚漏词；
4. 不对称提醒：反向平均（对 y 每词找 x）会得到不同值——实体对齐里常双向取 max 或配 blocking 键使用；
5. 本质：内层任何字符级度量（编辑/Jaro）都能插进外层词对齐框架——**度量可组合**。

```java [Java]
// Hamming：仅适用于等长串（定长编码、DNA motif、CRC 分析）
int hamming(String a, String b) {
    if (a.length() != b.length()) throw new IllegalArgumentException("长度必须相等");
    int d = 0;
    for (int i = 0; i < a.length(); i++)
        if (a.charAt(i) != b.charAt(i)) d++;
    return d;
}

// Q-Gram 相似度：把串切成重叠 k-gram 的 multiset，比交并（Jaccard 的 k-gram 版）
double qGram(String a, String b, int k) {
    java.util.Map<String, Integer> ca = new HashMap<>(), cb = new HashMap<>();
    for (int i = 0; i + k <= a.length(); i++)
        ca.merge(a.substring(i, i + k), 1, Integer::sum);
    for (int i = 0; i + k <= b.length(); i++)
        cb.merge(b.substring(i, i + k), 1, Integer::sum);
    long inter = 0, union = 0;
    for (var e : ca.entrySet()) {
        int c = Math.min(e.getValue(), cb.getOrDefault(e.getKey(), 0));
        inter += c;
        union += Math.max(e.getValue(), cb.getOrDefault(e.getKey(), 0));
    }
    for (var e : cb.entrySet())
        if (!ca.containsKey(e.getKey())) union += e.getValue();
    long totalA = Math.max(a.length() - k + 1, 0), totalB = Math.max(b.length() - k + 1, 0);
    long denom = Math.max(totalA + totalB - inter, 1);
    return 2.0 * inter / denom;             // Dice 式：|A∩B| 用 min-count，配 2|A∩B|/(|A|+|B|)
}
```

```python [Python]
def qgram_similarity(a, b, k=2):
    from collections import Counter
    ca = Counter(a[i:i + k] for i in range(len(a) - k + 1))
    cb = Counter(b[i:i + k] for i in range(len(b) - k + 1))
    inter = sum((ca & cb).values())          # Counter 交集 = 逐键取 min
    na, nb = sum(ca.values()), sum(cb.values())
    if na + nb == 0:
        return 1.0
    return 2 * inter / (na + nb)             # Dice 系数形态，值域 [0,1]
```

```rust [Rust]
pub fn qgram_dice(a: &str, b: &str, k: usize) -> f64 {
    use std::collections::HashMap;
    let grams = |s: &str| -> HashMap<String, usize> {
        let v: Vec<char> = s.chars().collect();
        let mut m = HashMap::new();
        if v.len() >= k {
            for i in 0..=v.len() - k {
                *m.entry(v[i..i + k].iter().collect()).or_default() += 1;
            }
        }
        m
    };
    let (ca, cb) = (grams(a), grams(b));
    let inter: usize = ca.iter()
        .map(|(g, c)| (*c).min(*cb.get(g).unwrap_or(&0)))
        .sum();
    let na: usize = ca.values().sum();
    let nb: usize = cb.values().sum();
    if na + nb == 0 { return 1.0; }
    2.0 * inter as f64 / (na + nb) as f64
}
```

```typescript [TypeScript]
  if (a.length !== b.length) throw new Error("长度必须相等");
  let d = 0;
  for (let i = 0; i < a.length; i++) if (a[i] !== b[i]) d++;
  return d;
}

function qgramDice(a: string, b: string, k = 2): number {
  const grams = (s: string) => {
    const m = new Map<string, number>();
    for (let i = 0; i + k <= s.length; i++) {
      const g = s.slice(i, i + k);
      m.set(g, (m.get(g) ?? 0) + 1);
    }
    return m;
  };
  const ca = grams(a), cb = grams(b);
  let inter = 0, na = 0, nb = 0;
  for (const c of ca.values()) na += c;
  for (const c of cb.values()) nb += c;
  for (const [g, c] of ca) inter += Math.min(c, cb.get(g) ?? 0);
  return na + nb === 0 ? 1 : (2 * inter) / (na + nb);
}
```

## Jaro / Jaro-Winkler

::: code-group

```java [Java]
double jaro(String a, String b) {
    int n = a.length(), m = b.length();
    if (n == 0 && m == 0) return 1.0;
    if (n == 0 || m == 0) return 0.0;
    int win = Math.max(n, m) / 2 - 1;
    if (win < 0) win = 0;
    boolean[] usedB = new boolean[m];
    int[] pairA = new int[n];
    java.util.Arrays.fill(pairA, -1);
    int matches = 0;
    for (int i = 0; i < n; i++) {                 // ① 窗口内贪心配对
        int lo = Math.max(0, i - win), hi = Math.min(m - 1, i + win);
        for (int j = lo; j <= hi; j++) {
            if (!usedB[j] && a.charAt(i) == b.charAt(j)) {
                usedB[j] = true; pairA[i] = j; matches++;
                break;
            }
        }
    }
    if (matches == 0) return 0.0;
    int transpositions = 0;                       // ② 配好序列的错位对数 /2
    int k = 0;
    for (int i = 0; i < n; i++) {
        if (pairA[i] < 0) continue;
        while (!usedB[k] || pairA[i] != k) k++;   // 定位 b 中第 k 个匹配位
        if (a.charAt(i) != b.charAt(k)) transpositions++;
    }
    transpositions /= 2;
    return (matches / (double) n + matches / (double) m
            + (matches - transpositions) / (double) matches) / 3.0;
}

double jaroWinkler(String a, String b, double prefixScale) {  // 常用 0.1
    double j = jaro(a, b);
    int prefix = 0;
    while (prefix < 4 && prefix < Math.min(a.length(), b.length())
           && a.charAt(prefix) == b.charAt(prefix)) prefix++;
    return j + prefixScale * prefix * (1 - j);
}
```

```python [Python]
def jaro(a, b):
    n, m = len(a), len(b)
    if n == m == 0:
        return 1.0
    if n == 0 or m == 0:
        return 0.0
    win = max(0, max(n, m) // 2 - 1)
    used_b, pairs = [False] * m, []
    for i in range(n):                            # ① 窗口配对
        for j in range(max(0, i - win), min(m, i + win + 1)):
            if not used_b[j] and a[i] == b[j]:
                used_b[j] = True
                pairs.append((i, j))
                break
    mt = len(pairs)
    if mt == 0:
        return 0.0
    seq_a = [a[i] for i, _ in pairs]              # ② 按序抽出数换位
    seq_b = [b[j] for _, j in pairs]
    t = sum(1 for x, y in zip(seq_a, seq_b) if x != y) // 2
    return (mt / n + mt / m + (mt - t) / mt) / 3

def jaro_winkler(a, b, scale=0.1):
    j = jaro(a, b)
    l = next((i for i in range(min(4, min(len(a), len(b))))
              if a[i] != b[i]), min(4, min(len(a), len(b))))
    return j + scale * l * (1 - j)
```

```rust [Rust]
pub fn jaro(a: &str, b: &str) -> f64 {
    let (va, vb): (Vec<char>, Vec<char>) = (a.chars().collect(), b.chars().collect());
    let (n, m) = (va.len(), vb.len());
    if n == 0 && m == 0 { return 1.0; }
    if n == 0 || m == 0 { return 0.0; }
    let win = (n.max(m) / 2).saturating_sub(1);
    let mut used = vec![false; m];
    let mut pairs: Vec<(usize, usize)> = Vec::new();
    for i in 0..n {
        for j in i.saturating_sub(win)..=((i + win).min(m - 1)) {
            if !used[j] && va[i] == vb[j] {
                used[j] = true;
                pairs.push((i, j));
                break;
            }
        }
    }
    let mt = pairs.len();
    if mt == 0 { return 0.0; }
    let t = (0..mt)
        .filter(|&k| va[pairs[k].0] != vb[pairs[k].1])
        .count() / 2;
    (mt as f64 / n as f64 + mt as f64 / m as f64
        + (mt - t) as f64 / mt as f64) / 3.0
}

pub fn jaro_winkler(a: &str, b: &str, scale: f64) -> f64 {
    let j = jaro(a, b);
    let l = a.chars().zip(b.chars())
        .take(4)
        .take_while(|(x, y)| x == y)
        .count();
    j + scale * l as f64 * (1.0 - j)
}
```

```typescript [TypeScript]
function jaro(a: string, b: string): number {
  const n = a.length, m = b.length;
  if (n === 0 && m === 0) return 1;
  if (n === 0 || m === 0) return 0;
  const win = Math.max(0, Math.floor(Math.max(n, m) / 2) - 1);
  const used = new Array(m).fill(false);
  const pairs: [number, number][] = [];
  for (let i = 0; i < n; i++)
    for (let j = Math.max(0, i - win); j <= Math.min(m - 1, i + win); j++)
      if (!used[j] && a[i] === b[j]) {
        used[j] = true; pairs.push([i, j]); break;
      }
  const mt = pairs.length;
  if (mt === 0) return 0;
  let t = 0;
  for (let k = 0; k < mt; k++) if (a[pairs[k][0]] !== b[pairs[k][1]]) t++;
  t = Math.floor(t / 2);
  return (mt / n + mt / m + (mt - t) / mt) / 3;
}

function jaroWinkler(a: string, b: string, scale = 0.1): number {
  const j = jaro(a, b);
  let l = 0;
  while (l < 4 && l < Math.min(a.length, b.length) && a[l] === b[l]) l++;
  return j + scale * l * (1 - j);
}
```

## Monge-Elkan 与 Soundex

::: code-group

```java [Java]
// Monge-Elkan：外层词对齐，内层任取字符级相似度 sim（此处用归一化 Levenshtein）
double mongeElkan(String x, String y, java.util.function.BiFunction<String, String, Double> sim) {
    String[] xs = x.trim().split("\\s+"), ys = y.trim().split("\\s+");
    double sum = 0;
    for (String w : xs) {
        double best = 0;
        for (String v : ys) best = Math.max(best, sim.apply(w, v));
        sum += best;                             // ys 为空时 best 恒 0，天然惩罚
    }
    return sum / xs.length;
}

// Soundex：美国姓名音码，首字母 + 3 位数字（同组辅音同码）
String soundex(String name) {
    String s = name.toUpperCase().replaceAll("[^A-Z]", "");
    if (s.isEmpty()) return "";
    StringBuilder out = new StringBuilder().append(s.charAt(0));
    char prev = code(s.charAt(0));               // 首字母的码参与「连续同码去重」
    for (int i = 1; i < s.length() && out.length() < 4; i++) {
        char ch = s.charAt(i);
        char c = code(ch);
        if (c != '0' && c != prev) out.append(c);
        if (ch != 'H' && ch != 'W') prev = c;    // H/W 不隔断同码，元音（码 0）隔断
    }
    while (out.length() < 4) out.append('0');
    return out.toString();
}
char code(char ch) {                             // 标准分组：BFPV=1 CGJKQSXZ=2 DT=3 L=4 MN=5 R=6
    if ("BFPV".indexOf(ch) >= 0) return '1';
    if ("CGJKQSXZ".indexOf(ch) >= 0) return '2';
    if ("DT".indexOf(ch) >= 0) return '3';
    if (ch == 'L') return '4';
    if ("MN".indexOf(ch) >= 0) return '5';
    if (ch == 'R') return '6';
    return '0';
}
```

```python [Python]
def monge_elkan(x, y, char_sim):
    """char_sim: 词级相似度函数（如 1 - lev/max_len）"""
    xs, ys = x.split(), y.split()
    if not xs:
        return 1.0 if not ys else 0.0
    if not ys:
        return 0.0
    return sum(max(char_sim(w, v) for v in ys) for w in xs) / len(xs)

def soundex(name):
    """首字母 + 3 位辅音音码：BFPV=1 CGJKQSXZ=2 DT=3 L=4 MN=5 R=6"""
    s = "".join(c for c in name.upper() if c.isalpha())
    if not s:
        return ""
    table = {**dict.fromkeys("BFPV", "1"), **dict.fromkeys("CGJKQSXZ", "2"),
             **dict.fromkeys("DT", "3"), "L": "4",
             **dict.fromkeys("MN", "5"), "R": "6"}
    def code(ch):
        return table.get(ch, "0")
    out = [s[0]]
    prev = code(s[0])
    for ch in s[1:]:
        c = code(ch)
        if c != "0" and c != prev:
            out.append(c)
        prev = c if ch not in "HW" else prev   # H/W 不隔断同码，AEIOUY 隔断
        if len(out) == 4:
            break
    return "".join(out).ljust(4, "0")
```

```rust [Rust]
pub fn monge_elkan<F: Fn(&str, &str) -> f64>(x: &str, y: &str, sim: F) -> f64 {
    let xs: Vec<&str> = x.split_whitespace().collect();
    let ys: Vec<&str> = y.split_whitespace().collect();
    if xs.is_empty() { return if ys.is_empty() { 1.0 } else { 0.0 }; }
    if ys.is_empty() { return 0.0; }
    xs.iter()
        .map(|w| ys.iter().map(|v| sim(w, v)).fold(0.0, f64::max))
        .sum::<f64>() / xs.len() as f64
}

pub fn soundex(name: &str) -> String {
    let s: Vec<char> = name.to_uppercase()
        .chars().filter(|c| c.is_ascii_alphabetic()).collect();
    if s.is_empty() { return String::new(); }
    let code = |c: char| match c {
        'B' | 'F' | 'P' | 'V' => '1',
        'C' | 'G' | 'J' | 'K' | 'Q' | 'S' | 'X' | 'Z' => '2',
        'D' | 'T' => '3',
        'L' => '4',
        'M' | 'N' => '5',
        'R' => '6',
        _ => '0',
    };
    let mut out = String::from(s[0]);
    let mut prev = code(s[0]);
    for &ch in &s[1..] {
        let c = code(ch);
        if c != '0' && c != prev {
            out.push(c);
            if out.len() == 4 { break; }
        }
        if !matches!(ch, 'H' | 'W') { prev = c; }  // AEIOUY 视同 0 会重置 prev
    }
    while out.len() < 4 { out.push('0'); }
    out
}
```

```typescript [TypeScript]
function mongeElkan(x: string, y: string,
                    sim: (a: string, b: string) => number): number {
  const xs = x.trim().split(/\s+/), ys = y.trim().split(/\s+/);
  if (!xs[0]) return ys.length ? 0 : 1;
  if (!ys[0]) return 0;
  let sum = 0;
  for (const w of xs) {
    let best = 0;
    for (const v of ys) best = Math.max(best, sim(w, v));
    sum += best;
  }
  return sum / xs.length;
}

function soundex(name: string): string {
  const s = name.toUpperCase().replace(/[^A-Z]/g, "");
  if (!s) return "";
  const code = (c: string) =>
    "BFPV".includes(c) ? "1" : "CGJKQSXZ".includes(c) ? "2" :
    "DT".includes(c) ? "3" : c === "L" ? "4" : "MN".includes(c) ? "5" :
    c === "R" ? "6" : "0";
  let out = s[0], prev = code(s[0]);
  for (const ch of s.slice(1)) {
    const c = code(ch);
    if (c !== "0" && c !== prev) { out += c; if (out.length === 4) break; }
    if (!"HW".includes(ch)) prev = c;
  }
  return out.padEnd(4, "0");
}
```

## 六种度量怎么选

| 度量 | 适合 | 不适合 | 典型阈值 |
|------|------|--------|---------|
| Hamming | 等长码比对（CRC/DNA 定长 motif） | 任何含插入/删除的串 | — |
| Jaro | 短人名/地名字符级 | 长文本（O(nm) 且窗口假设失效） | ≥0.8 |
| Jaro-Winkler | **姓名**（前缀敏感）、卡片验真 | 前后缀都不占优的通用串 | 0.85（人口普查） |
| Q-Gram | 长度悬殊串的粗筛、可布隆化海量去重 | 精确对齐（对位移不敏感） | 0.6~0.8 |
| Monge-Elkan | **地址**、公司名等词组结构串 | 无空格结构的单词 | 0.9 |
| Soundex | 英语姓名**音近**归一（`Smith/ Smyth` 同码） | 非英语姓名（中文拼音失真）、短码碰撞高 | 等码即候选 |

- 现代继承者：`python-Levenshtein`/`rapidfuzz`（token_sort_ratio≈ME 思想）、Metaphone/Double Metaphone（Soundex 的音素学加强版）、LSH+Q-Gram（海量去重）。
- 工程组合拳：**Blocking**（Soundex/Q-Gram 粗筛出候选对）→ **Scoring**（ME/JW 精算）→ 阈值+规则裁决——百万级记录对齐从不两两全比。

## 常见陷阱

- Hamming 对不等长串直接非法——要么先截断/补齐（谨慎声明），要么改用 Levenshtein。
- Jaro 的换位定义是「配对序列中不一致的对数 **÷2**」，忘除 2 或把未配对字符卷进比较都会偏低；窗口 `max/2−1` 的整除行为各语言一致但注意 `n=1` 时为 0。
- Jaro-Winkler 前缀奖励可把分数推到 >1 的错觉——scale 不超 `1/4` 才保证 ≤1（惯例 0.1）。
- Monge-Elkan 不对称：`ME(x,y) ≠ ME(y,x)`（对 x 每词取 max 再平均）；查重时固定「短边为 x」或双向取 max。
- Soundex 只有 4 位码，碰撞率极高——它是**分块键**不是判据，等码后必须再跑字符级度量。
- Q-Gram 的 k 值敏感：k=2 宽松、k=4 苛刻，跨语言/数据集要重新标定阈值。

# 总结

- 六件套各守一层：Hamming 管等长、Jaro/JW 管短名字形、Q-Gram 管任意长度的片重合、ME 管词组结构（内层随便插什么字符度量）、Soundex 管发音——先问数据长什么样，再挑尺子；
- 相似度分数没有绝对意义，**阈值才是产品决策**：0.85 收、0.7 人审、其余放行的三段式是实体对齐常态；
- 大规模对齐的架构记忆：**粗筛保召回（音码/指纹）、精算保准确（编辑/ME）**——单度量包打天下是伪命题。
