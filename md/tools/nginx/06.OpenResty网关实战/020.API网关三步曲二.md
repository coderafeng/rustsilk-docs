# 概述

**Step 2**：在 Step 1 的路由之上加**鉴权 + 限流**。核心：**JWT 校验 + API Key 双通道**、**多维令牌桶 + 白名单**、**全部走 shdict 配置热更**。

# 目标架构增量

```text
Client ─▶ gateway
              │
              ├─ access_by_lua  ← ★ Step 2 新加
              │    ├─ authn：JWT / API Key
              │    └─ ratelimit：tenant + user + api 三维
              ├─ rewrite_by_lua：路径匹配（Step 1）
              ├─ proxy_pass
              └─ balancer_by_lua（Step 1）
```

**放在 access 阶段的原因**：Step 1 的路由在 rewrite 里已经算好 `ngx.ctx.service`，access 阶段能拿到；早于任何 IO；拒绝请求成本最低。

# 配置结构（Consul KV）

```json
{
  "auth": {
    "jwt_secret": "${VAULT_JWT_SECRET}",
    "jwt_issuer": "auth.acme.com",
    "api_keys": {
      "partner_xxx_key": "tenant-1",

  },
  "ratelimit": {
    "rules": [
      { "key": "tenant:*",       "rate": 10000, "capacity": 20000 },
      { "key": "user:*",         "rate": 100,   "capacity": 200 },
      { "key": "api:/api/orders","rate": 500,   "capacity": 1000 }

```

**`jwt_secret` 从 Vault 注入**：不要写死配置；同步器启动时替换占位符。

# 鉴权模块

```lua
-- lua/auth.lua
local jwt  = require "resty.jwt"      -- lua-resty-jwt
local cjson = require "cjson.safe"

local sh_cfg = ngx.shared.routes     -- 复用 Step 1 的 dict；生产建议拆独立 dict
local sh_keys = ngx.shared.api_keys

local _M = {}

local function whitelist_hit(path)
    local raw = sh_cfg:get("auth.whitelist") or ""
    for pat in raw:gmatch("[^\n]+") do
        if ngx.re.find(path, "^" .. pat .. "$", "jo") then return true end
    end
    return false
end

function _M.check()
    local path = ngx.var.uri
    if whitelist_hit(path) then
        ngx.ctx.authenticated = "public"
        return true
    end

    local hdr = ngx.req.get_headers(50)
    local auth = hdr["authorization"] or ""

    -- 优先 JWT
    local bearer = auth:match("^Bearer%s+(.+)$")
    if bearer then
        local verified = jwt:verify(sh_cfg:get("auth.jwt_secret"), bearer)
        if not verified.verified then
            return nil, verified.reason or "invalid_token"
        end
        local payload = verified.payload
        if payload.exp and payload.exp < ngx.now() then
            return nil, "token_expired"
        end
        ngx.ctx.user_id  = payload.sub
        ngx.ctx.tenant   = payload.tenant
        ngx.ctx.scopes   = payload.scopes or {}
        ngx.ctx.authenticated = "jwt"
        return true
    end

    -- 兜底 API Key
    local key = hdr["x-api-key"]
    if key then
        local tenant = sh_keys:get(key)
        if not tenant then return nil, "invalid_api_key" end
        ngx.ctx.tenant = tenant
        ngx.ctx.user_id = "api_key:" .. tenant
        ngx.ctx.authenticated = "api_key"
        return true
    end

    return nil, "missing_credential"
end

return _M
```

**要点**：
- `resty.jwt` 走 LuaJIT + OpenSSL FFI，一次验签 <50μs；
- API Key 直接 shdict hash 查（key 是 API Key 字符串，value 是 tenant）；
- `ngx.ctx` 存 `authenticated`，方便限流与日志区分。

# 限流模块

```lua
-- lua/ratelimit.lua
local lock = require "resty.lock"
local cjson = require "cjson.safe"

local sh_state = ngx.shared.metrics   -- 生产换独立 dict
local sh_cfg   = ngx.shared.routes
local locker   = lock:new("locks", { timeout = 3 })

local _M = {}

local function rules()
    local raw = sh_cfg:get("ratelimit.rules")
    if not raw then return {} end
    return cjson.decode(raw) or {}
end

-- 令牌桶：state 存在 shdict；每次 refill
local function bucket(key, rate, capacity)
    local now_ms = ngx.now() * 1000
    local tokens = sh_state:get("rl:" .. key)
    if not tokens then tokens = capacity end
    local last_ms = sh_state:get("rl_ts:" .. key) or now_ms
    tokens = math.min(capacity, tokens + (now_ms - last_ms) / 1000 * rate)

    if tokens < 1 then
        sh_state:set("rl:" .. key, tokens, 300)
        sh_state:set("rl_ts:" .. key, now_ms, 300)
        return false
    end

    tokens = tokens - 1
    sh_state:set("rl:" .. key, tokens, 300)
    sh_state:set("rl_ts:" .. key, now_ms, 300)
    return true
end

function _M.check()
    for _, r in ipairs(rules()) do
        local k = _M.resolve_key(r.key)
        if k then
            local ok = bucket(k, r.rate, r.capacity or r.rate * 2)
            if not ok then
                sh_state:incr("rl_reject:" .. r.key, 1, 0)
                return false, r
            end
            sh_state:incr("rl_pass:" .. r.key, 1, 0)
        end
    end
    return true
end

function _M.resolve_key(pattern)
    -- pattern 形如 "tenant:*" / "user:*" / "api:/api/orders"
    local kind, val = pattern:match("^(%w+):(.+)$")
    if kind == "tenant" and ngx.ctx.tenant then
        return "t:" .. ngx.ctx.tenant
    elseif kind == "user" and ngx.ctx.user_id then
        return "u:" .. ngx.ctx.user_id
    elseif kind == "api" then
        return "a:" .. val
    end
end

return _M
```

# 在 nginx.conf 里接入

```nginx
location / {
    # Step 1 的路由匹配（rewrite）
    rewrite_by_lua_block {
        local router = require "router"
        local svc = router.match(ngx.var.uri)
        if not svc then return ngx.exit(404) end
        ngx.ctx.service = svc
    }

    # ★ Step 2 新增：鉴权 + 限流
    access_by_lua_block {
        local auth = require "auth"
        local ok, err = auth.check()
        if not ok then
            ngx.status = 401
            ngx.header["WWW-Authenticate"] = 'Bearer realm="gateway"'
            ngx.header["Content-Type"] = "application/json"
            return ngx.print({ code = 401, msg = err })
        end

        local rl = require "ratelimit"
        local pass, rule = rl.check()
        if not pass then
            ngx.status = 429
            ngx.header["Retry-After"] = "1"
            ngx.header["Content-Type"] = "application/json"
            return ngx.print(rule.reject_body or { code = 429 })
        end
    }

    proxy_pass http://internal_backend;
    # ... Step 1 的配置
}
```

# 同步器扩展：把 auth / rl 配置也拉进来

```lua
-- lua/syncer.lua（Step 1 基础上）
local function pull_gateway_config()
    local res = http.request_uri(CONSUL .. "/v1/kv/gateway/config?raw")
    if not res or res.status ~= 200 then return end
    local cfg = cjson.decode(res.body) or {}

    if cfg.auth then
        sh_cfg:set("auth.jwt_secret", cfg.auth.jwt_secret, 30)
        sh_cfg:set("auth.whitelist", table.concat(cfg.auth.whitelist_paths or {}, "\n"), 30)
        for k, v in pairs(cfg.auth.api_keys or {}) do
            ngx.shared.api_keys:set(k, v, 30)
        end
    end
    if cfg.ratelimit then
        sh_cfg:set("ratelimit.rules", cjson.encode(cfg.ratelimit.rules or {}), 30)
    end
end

function _M.start()
    ngx.timer.every(10, function(prem)
        if prem then return end
        pull_gateway_config()
        -- Step 1 的 routes/upstreams 同步
    end)
end
```

**关键**：所有配置都 TTL 30s；Consul 挂时仍能用最后一次数据；30s 后自动过期，避免陈旧。

# 交付验证清单

- [ ] 无 token → 401；错误 token → 401；正确 JWT → 放行；
- [ ] API Key 命中 → 放行；未知 → 401；
- [ ] 白名单 path 直接放行；
- [ ] 单 tenant 打满 10000 QPS 时超出部分 429；
- [ ] Consul 里改 rate，15s 内新阈值生效；
- [ ] `/metrics` 有 auth_* 与 rl_* 计数；
- [ ] wrk 用同一 token 打 200 QPS 1 分钟无 5xx。

# 避坑

- **JWT secret 别放 config 明文**：Vault / KMS；同步器拉；
- **API Key shdict 是全局**：`api_keys` dict 权限收紧（admin only 写）；
- **`get_keys` 遍历 metrics 会 stall**：只给 `/metrics` 端点用；控制上限；
- **限流 key 爆炸**：`user_id:*` 每用户一个 key；上百万用户时 shdict 撑不住——按 tenant 粗粒度 + Redis 兜底；
- **`set_more_tries` 与 429 冲突**：`balancer` 里 retry 只处理连接错误；429 是网关直接返；不要混淆。

# 总结

- Step 2 = Step 1 + JWT/API Key + 多维令牌桶；
- 鉴权在 access 早于任何 IO；
- 所有配置走 Consul → shdict，15s 生效；
- 指标齐全：auth / rl 两类；
- 到这里网关可以对外开放。
