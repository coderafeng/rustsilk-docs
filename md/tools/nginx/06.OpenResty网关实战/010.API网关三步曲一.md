# 概述

**Step 1**：从"什么网关都没有"到一个能自动同步服务列表、动态转发的最小网关。核心组件：**服务发现 + 路由表 shdict + `balancer_by_lua` + `access_by_lua`**。

# 目标架构

```text
Client ─▶ gateway :8080
              │
              ├─ access_by_lua: 根据 path 匹配路由，取服务名
              ├─ proxy_pass: 转到内部 location
              ├─ balancer_by_lua: 从 shdict 里挑一个健康实例
              └─ 转发到实际上游
              ┌────────────┐
              │ Consul     │◀── timer 每 10s 拉服务列表
              └────────────┘
              │
              ├─ shdict: routes  (服务 → 路径映射)
              └─ shdict: upstreams (服务 → 实例列表)
```

# 完整 nginx.conf

```nginx
worker_processes auto;
error_log logs/error.log warn;
pid logs/nginx.pid;

events { worker_connections 10240; }

http {
    lua_package_path "$prefix/lua/?.lua;;";

    lua_shared_dict routes    4m;
    lua_shared_dict upstreams 8m;
    lua_shared_dict locks     1m;
    lua_shared_dict metrics   2m;

    lua_code_cache on;
    lua_socket_pool_size 64;
    resolver 127.0.0.1 valid=10s;
    resolver_timeout 3s;

    init_by_lua_block {
        require "resty.core"
    }

    init_worker_by_lua_block {
        local syncer = require "syncer"
        syncer.start()
    }

    server {
        listen 8080 reuseport;
        server_name _;

        # 主入口
        location / {
            access_by_lua_block {
                local router = require "router"
                local svc = router.match(ngx.var.uri)
                if not svc then
                    return ngx.exit(404)
                end
                ngx.ctx.service = svc
            }
            proxy_pass http://internal_backend;
            proxy_http_version 1.1;
            proxy_set_header Connection "";
            proxy_set_header Host $upstream_host;
            proxy_set_header X-Trace-Id $request_id;
        }

        # 内部 upstream 由 balancer 阶段动态指定
        upstream internal_backend {
            server 0.0.0.1:1;                  -- 占位；真地址在 balancer 里给
            balancer_by_lua_block {
                local lb = require "lb"
                lb.pick(ngx.ctx.service)
            }
            keepalive 512;
        }

        location = /healthz { return 200 "ok\n"; }
    }

    server {
        listen 9090;
        location = /metrics {
            content_by_lua_block {
                local metrics = require "metrics"
                metrics.dump()
            }
        }
    }
}
```

**要点**：
- `upstream { server 0.0.0.1:1; }` 是**占位**，真实地址在 `balancer_by_lua` 里给；
- `keepalive 512` 让每个服务共用一个连接池；
- `resolver` 让 cosocket 里的域名能解析；
- `init_worker_by_lua` 起后台 timer 从 Consul 同步。

# 路由模块

```lua
-- lua/router.lua
local sh_routes = ngx.shared.routes
local cjson = require "cjson.safe"
local lrucache = require "resty.lrucache"

local _M = {}
local l1, _ = lrucache.new(200)

function _M.match(uri)
    local svc = l1:get(uri)
    if svc ~= nil then return svc or nil end

    local raw = sh_routes:get("table")
    if not raw then l1:set(uri, false, 1); return nil end
    local rules = cjson.decode(raw) or {}

    for _, r in ipairs(rules) do
        local from = ngx.re.find(uri, r.pattern, "jo")
        if from then
            l1:set(uri, r.service, 5)
            return r.service
        end
    end
    l1:set(uri, false, 1)
    return nil
end

return _M
```

**规则**（存在 Consul KV）：

```json
[
  { "pattern": "^/api/orders/?",   "service": "orders-svc" },
  { "pattern": "^/api/users/\\d+$","service": "users-svc" },
  { "pattern": "^/api/pay/?",      "service": "pay-svc" }
]
```

**L1 5s + shdict 秒级**：路由改动 <15s 生效。

# 服务发现同步

```lua
-- lua/syncer.lua
local http = require "resty.http"
local cjson = require "cjson.safe"

local _M = {}

local CONSUL = "http://consul.local:8500"

local function fetch_routes()
    local res = http.request_uri(CONSUL .. "/v1/kv/gateway/routes?raw")
    if res and res.status == 200 then return res.body end
end

local function fetch_upstream(name)
    local res = http.request_uri(
        CONSUL .. "/v1/health/service/" .. name .. "?passing")
    if res and res.status == 200 then
        local list = {}
        for _, svc in ipairs(cjson.decode(res.body) or {}) do
            local n = svc.Node
            local s = svc.Service
            list[#list + 1] = { host = n.Address, port = s.Port }
        end
        return cjson.encode(list)
    end
end

function _M.start()
    ngx.timer.every(10, function(prem)
        if prem then return end
        local raw = fetch_routes()
        if raw then ngx.shared.routes:set("table", raw, 30) end

        local routes = cjson.decode(raw or "[]") or {}
        local seen = {}
        for _, r in ipairs(routes) do
            if not seen[r.service] then
                seen[r.service] = true
                local ups = fetch_upstream(r.service)
                if ups then ngx.shared.upstreams:set(r.service, ups, 30) end
            end
        end
    end)
end

return _M
```

**要点**：
- `timer.every(10, ...)` 只让**抢到 shdict 锁的 worker** 干；见 [03 章 060](../03.OpenResty核心API/060.特权进程与定时任务.md)；
- 简化版没加锁——生产补上：`sh_locks:add("sync_lock", pid, 5)`；
- 每次 TTL 30s：即使 Consul 挂了，30s 内还能读到旧数据。

# 负载均衡

```lua
-- lua/lb.lua
local balancer = require "ngx.balancer"
local cjson = require "cjson.safe"

local _M = {}
local idx_cache = setmetatable({}, { __mode = "k" })   -- service → 当前索引

function _M.pick(service)
    if not service then return ngx.exit(502) end

    local raw = ngx.shared.upstreams:get(service)
    if not raw then
        ngx.log(ngx.WARN, "no upstream for ", service)
        return ngx.exit(503)
    end

    local nodes = cjson.decode(raw)
    if not nodes or #nodes == 0 then return ngx.exit(503) end

    -- 简单轮询
    local idx = (idx_cache[service] or 0) % #nodes + 1
    idx_cache[service] = idx
    local node = nodes[idx]

    local ok, err = balancer.set_current_peer(node.host, node.port)
    if not ok then
        ngx.log(ngx.ERR, "set_current_peer failed: ", err)
        return ngx.exit(502)
    end
    balancer.set_more_tries(1)               -- 失败重试 1 次
end

return _M
```

**要点**：
- `set_current_peer` 只在 `balancer_by_lua_block` 里可用（见 [01 章 070](../01.OpenResty起步/070.请求处理阶段总览.md) 矩阵）；
- `set_more_tries` 让下一次 retry 会重新进 `balancer` → 可以换节点；
- 生产换更聪明的算法：一致性哈希 / 最少连接 / P2C。

# 避坑

- **`proxy_pass` 里的 `$internal_backend` 别写死**：走 `upstream` 块，让 `balancer_by_lua` 起作用；
- **`resolver` 缺失**：Consul 里的实例名是 DNS 时就挂；给 IP 时无所谓；
- **`lua_code_cache off`**：开发时方便，生产一定 `on`；
- **`set_more_tries` 忘调**：502 直接返回给客户；重试一次能盖住大多数瞬时故障。

# 总结

- Step 1 完成：路径匹配 + 服务发现 + 动态 balancer 三件套；
- 关键模块：`router` / `syncer` / `lb` 三个 Lua 文件；
- shdict 三层：`routes` / `upstreams` / `locks`；
- 路由热更 <15s，实例上下线 <30s；
- 有了这一版，就能替代"手工维护 upstream 列表"的 nginx。
