# 概述

前两篇讲了缓存分层（090）和风暴治理（100）。工程上更实用的是**封装一个统一入口**：`cache.get(key, fetch_fn, opts)`——业务代码不用管命中在哪层、怎么加锁、怎么打散 TTL。本文给一个可直接抄的多级缓存封装。

# 设计目标

```lua
-- 业务代码只关心：怎么拿一个 key 的值
local user = cache.get("user:1", function()
    return db:query_user(1)
end, {
    l1_ttl = 5,
    l2_ttl = 60,
    neg_ttl = 30,
    soft_ttl = 20,
})
```

- API 简洁：一个 `get(key, fetch, opts)`；
- 内部自动走 L1 → L2 → 回源；
- 内置锁 + 打散 TTL + 负缓存 + serve-stale；
- 可观测：命中率、耗时、锁竞争。

# 骨架

```lua
-- lib/multi_cache.lua
local lrucache = require "resty.lrucache"
local lock     = require "resty.lock"

local _M = {}
local mt = { __index = _M }

function _M.new(opts)
    local l1, err = lrucache.new(opts.l1_size or 500)
    if not l1 then return nil, err end
    local locker = lock:new(opts.lock_dict or "locks", { timeout = 5 })
    return setmetatable({
        l1     = l1,
        l2     = ngx.shared[opts.shdict],
        locker = locker,
        opts   = opts,
        stats  = { hit_l1 = 0, hit_l2 = 0, miss = 0, fetched = 0 },
    }, mt)
end

local function jitter(ttl)
    return math.max(1, ttl + ngx.random(-ttl // 5, ttl // 5))
end

function _M:get(key, fetch, opts)
    opts = opts or self.opts
    local now = ngx.now()

    -- L1
    local v, stale_flag = self.l1:get(key)
    if v ~= nil then
        self.stats.hit_l1 = self.stats.hit_l1 + 1
        if stale_flag and stale_flag < now then
            self:_refresh_async(key, fetch, opts)
        end
        return v
    end

    -- L2
    v, stale_flag = self.l2:get(key)
    if v ~= nil then
        self.stats.hit_l2 = self.stats.hit_l2 + 1
        self.l1:set(key, v, opts.l1_ttl, stale_flag)
        if stale_flag and stale_flag < now then
            self:_refresh_async(key, fetch, opts)
        end
        return v
    end

    -- 负缓存短路
    if v == "__MISS__" then return nil end

    -- 回源 + 锁
    return self:_fetch_locked(key, fetch, opts)
end

function _M:_fetch_locked(key, fetch, opts)
    local elapsed, err = self.locker:lock("fetch:" .. key)
    if err == "timeout" then return nil, "busy" end

    -- double check
    local v = self.l2:get(key)
    if v then self.l1:set(key, v, opts.l1_ttl); self.locker:unlock(); return v end

    v = fetch()
    if v == nil or v == false then
        self.l2:set(key, "__MISS__", opts.neg_ttl or 30)
    else
        local soft_expire = ngx.now() + (opts.soft_ttl or opts.l2_ttl)
        self.l2:set(key, v, jitter((opts.l2_ttl or 60) * 2), soft_expire)
        self.l1:set(key, v, opts.l1_ttl, soft_expire)
    end
    self.locker:unlock()
    self.stats.fetched = self.stats.fetched + 1
    return v
end

function _M:_refresh_async(key, fetch, opts)
    if not self.l2:add("refreshing:" .. key, true, 1) then return end
    ngx.timer.at(0, function(prem)
        if prem then return end
        local ok, err = pcall(fetch)
        if ok then
            local fresh = fetch()
            if fresh ~= nil then
                local soft = ngx.now() + (opts.soft_ttl or opts.l2_ttl)
                self.l2:set(key, fresh, jitter((opts.l2_ttl or 60) * 2), soft)
            end
        end
    end)
end

function _M:stats()
    return self.stats
end

return _M
```

# 使用

```nginx
lua_shared_dict user_cache 32m;
lua_shared_dict locks      2m;
```

```lua
-- init_worker_by_lua_block
local multi_cache = require "multi_cache"
USER_CACHE = assert(multi_cache.new({
    shdict   = "user_cache",
    lock_dict = "locks",
    l1_size  = 500,
    l1_ttl   = 5,
    l2_ttl   = 60,
    soft_ttl = 20,
    neg_ttl  = 30,
}))

-- content_by_lua_block
local user = USER_CACHE:get("user:" .. id, function()
    local redis = require "resty.redis"
    local r = redis:new()
    r:connect("127.0.0.1", 6379)
    local data = r:get("user:" .. id)
    r:setkeepalive()
    return data and cjson.decode(data) or nil
end)
```

# 关键设计决策

## 1. `l2_ttl * 2` + `soft_ttl`

**物理 TTL = 逻辑 TTL × 2**：物理还没过期，但 `soft_ttl` 已过 → 异步 refresh 拿新数据。**用户永远拿到"能用"的值**，且总有一份在回源。

## 2. `jitter` 用 `ngx.random`

`ngx.random` 协程安全、每 worker 独立种子；`math.random` 在 LuaJIT 里跨协程串扰（见 [02 章 090](../02.Lua与OpenResty运行时内幕/090.OpenResty中Lua的约束与常见坑.md)）。

## 3. double check

拿到锁之后**再读一次 L2**——很可能上一个 owner 已经拉完。省 80% 回源。

## 4. refresh 去重

`self.l2:add("refreshing:" .. key, true, 1)` = "1 秒内同一 key 只起一次 refresh timer"。防止 serve-stale 下 refresh 爆炸。

# 观测埋点

```lua
-- /metrics
content_by_lua_block {
    local s = USER_CACHE:stats()
    local total = s.hit_l1 + s.hit_l2 + s.miss
    ngx.say("cache_hit_l1 ", s.hit_l1)
    ngx.say("cache_hit_l2 ", s.hit_l2)
    ngx.say("cache_fetched ", s.fetched)
    ngx.say("cache_hit_rate ", (s.hit_l1 + s.hit_l2) / math.max(1, total))
}
```

**关键**：hit 率、fetched 数、refresh timer 数三个指标进 Grafana，日常巡检。

# 避坑

- **`fetch` 里再递归 cache.get**：死锁；`fetch` 必须是"纯回源"；
- **`neg_ttl` 太长**：真的数据刚写入，负缓存还在，看不到；给 30s 内；
- **`l1_size` 太大**：worker 内存爆；根据热点 key 数调；
- **`shdict` 满了**：`set` 返回 `premature = true`，需要监控；见 03 章 040。

# 总结

- 一个 `cache.get(key, fetch, opts)` 承担 L1/L2/锁/负缓存/serve-stale；
- 物理 TTL × 2 + 逻辑 soft_ttl 是核心机制；
- jitter、double check、refresh 去重是三大稳定化技巧；
- 命中率与 refresh timer 数必须打指标。
